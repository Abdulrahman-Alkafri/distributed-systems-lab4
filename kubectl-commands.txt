 A. NGINX  - HTTPS, redirect, and hybrid load balancing
---------------------------------------------------------------------

---------------------------------------------------------------------
 C. SSE  - real-time updates without refreshing
---------------------------------------------------------------------

# C1. Open the live stream for 5 seconds. One request stays OPEN and the
#     server keeps pushing "data: {...}" lines.              [auto-stops]
timeout 5 curl -skN "https://$(minikube ip):30443/stream?cid=$(kubectl get pods -l app=flask -o jsonpath='{.items[0].metadata.name}')"

# C2. THE BEST DEMO: watch the number GO UP inside ONE open connection
#     while traffic is sent in the background. No new requests are made.
( sleep 2; for i in $(seq 1 6); do curl -sk -o /dev/null https://$(minikube ip):30443/; sleep 0.8; done ) & timeout 8 curl -skN "https://$(minikube ip):30443/stream?cid=$(kubectl get pods -l app=flask -o jsonpath='{.items[0].metadata.name}')"

#     Example output - notice "global" climbing with NO new request:
#       data: {"global": 139, "node": 43, "partition": "0"}
#       data: {"global": 141, "node": 44, "partition": "0"}
#       data: {"global": 143, "node": 44, "partition": "0"}
#       data: {"global": 145, "node": 45, "partition": "0"}


---------------------------------------------------------------------
 D. REDPANDA  - the 3-broker cluster (StatefulSet)
---------------------------------------------------------------------

# D1. The 3 broker pods with FIXED names redpanda-0 / -1 / -2
kubectl get pods -l app=redpanda

# D2. The cluster: 3 brokers (the one with * is the controller)
kubectl exec redpanda-0 -- rpk cluster info

# D3. Is it healthy? -> "Healthy: true"
kubectl exec redpanda-0 -- rpk cluster health

# D4. THE IMPORTANT ONE: 3 partitions, each replicated on ALL 3 brokers.
#     LEADER = which broker serves it.  REPLICAS [0 1 2] = copy on each broker.
#     EPOCH goes UP by 1 every time a new leader is elected.
kubectl exec redpanda-0 -- rpk topic describe visits-topic -p

# D5. All topics with partition and replica counts
kubectl exec redpanda-0 -- rpk topic list

# D6. Each broker has its OWN disk (PersistentVolumeClaim)
kubectl get pvc

# D7. WATCH MESSAGES ARRIVE LIVE - refresh the web page to see one appear
#                                                       [Ctrl+C to stop]
kubectl exec redpanda-0 -- rpk topic consume visits-topic -o end -f "P%p offset=%o key=%k value=%v\n"

# D8. KEY-BASED PARTITIONING PROOF: read ONLY partition 0 -> every message
#     has the same pod name as its key (that pod is pinned to this partition)
kubectl exec redpanda-0 -- rpk topic consume visits-topic -p 0 -o start -n 10 -f "P%p key=%k\n"

---------------------------------------------------------------------
 F. DLQ - ERROR HANDLING FOR BROKEN MESSAGES
    <<<<<  THE MAIN DEMO FOR THE PROFESSOR  >>>>>
---------------------------------------------------------------------

# STEP 1 - In TERMINAL A, watch the worker live:       [Ctrl+C to stop]
kubectl logs -f deployment/counter-worker

# STEP 2 - In TERMINAL B, send a message that is NOT valid JSON:
echo "this is not json" | kubectl exec -i redpanda-0 -- rpk topic produce visits-topic -k bad-1

# >>> Terminal A instantly prints:
#     [P1 @ offset 6] MALFORMED b'this is not json' -> sent to visits-dlq (JSONDecodeError(...))
#     and the worker KEEPS RUNNING - it does not crash.

# STEP 3 - Send valid JSON but with the WRONG field name:
echo '{"wrong_field": 1}' | kubectl exec -i redpanda-0 -- rpk topic produce visits-topic -k bad-2

# >>> Terminal A prints: MALFORMED ... (KeyError('container_id'))

# STEP 4 - Send JSON where container_id is empty:
echo '{"container_id": null}' | kubectl exec -i redpanda-0 -- rpk topic produce visits-topic -k bad-3

# STEP 5 - Show all 3 broken messages now parked in the DLQ topic,
#          each one WITH its error and where it came from:
kubectl exec redpanda-0 -- rpk topic consume visits-dlq -o start -n 3 -f "%k | %v | %h{%k=%v; }\n"

# >>> prints:
#     bad-1 | this is not json       | error=JSONDecodeError(...); source=visits-topic/1/6;
#     bad-2 | {"wrong_field": 1}     | error=KeyError('container_id'); source=visits-topic/1/7;
#     bad-3 | {"container_id": null} | error=KeyError('container_id'); source=visits-topic/2/0;

# STEP 6 - PROVE THE WORKER SURVIVED: RESTARTS must still be 0
kubectl get pods -l app=counter-worker

# STEP 7 - PROVE broken messages were NOT counted (only good ones count)
kubectl exec deployment/redis -- redis-cli get global_visits

# STEP 8 - Show every malformed message the worker has ever caught
kubectl logs deployment/counter-worker | grep MALFORMED

 H. PARTITION LOCKING - one partition belongs to ONE worker
---------------------------------------------------------------------

# H1. Run 3 counter workers instead of 1

# H2. Wait ~20 seconds, then look at MEMBER-ID:
#     3 DIFFERENT ids own P0, P1 and P2 - each partition is locked to one worker

# H4. A 4th worker would sit IDLE - only 3 partitions exist to hand out
