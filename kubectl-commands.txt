---------------------------------------------------------------------
 6. REDPANDA / KAFKA INSIDE KUBERNETES
---------------------------------------------------------------------
# Same rpk commands as Lab 3, just run inside a pod with "kubectl exec".

# The 3 brokers (the one with * is the controller)
kubectl exec redpanda-0 -- rpk cluster info

# Is the cluster healthy?
kubectl exec redpanda-0 -- rpk cluster health

# Topics with partitions and replicas
kubectl exec redpanda-0 -- rpk topic list

# Partitions of visits-topic:
#   LEADER  = which broker serves it
#   REPLICAS = [0 1 2] means a copy on all 3 brokers
#   EPOCH   = goes UP by 1 every time a new leader is elected
kubectl exec redpanda-0 -- rpk topic describe visits-topic -p

# The two consumer groups (fan-out)
kubectl exec redpanda-0 -- rpk group list

# Offsets, lag and which worker owns which partition
kubectl exec redpanda-0 -- rpk group describe counter-group
kubectl exec redpanda-0 -- rpk group describe audit-group

# WATCH MESSAGES ARRIVE LIVE (refresh the web page to see one appear) - Ctrl+C to stop
kubectl exec redpanda-0 -- rpk topic consume visits-topic -o end -f "P%p offset=%o key=%k value=%v\n"

# First 10 messages from the beginning
kubectl exec redpanda-0 -- rpk topic consume visits-topic -o start -n 10 -f "P%p offset=%o key=%k\n"

# Only partition 0 - proves one pod's events all land in the same partition
kubectl exec redpanda-0 -- rpk topic consume visits-topic -p 0 -o start -n 10 -f "P%p key=%k\n"

 7. DLQ TEST  <<< COPY-PASTE THIS IN FRONT OF THE PROFESSOR >>>
---------------------------------------------------------------------

# STEP 1 - in terminal A, watch the worker live:
kubectl logs -f deployment/counter-worker

# STEP 2 - in terminal B, send a BROKEN message (not valid JSON):
echo "this is not json" | kubectl exec -i redpanda-0 -- rpk topic produce visits-topic -k bad-1

# Terminal A immediately prints:
#   [P1 @ offset 6] MALFORMED b'this is not json' -> sent to visits-dlq (JSONDecodeError(...))
# and the worker KEEPS RUNNING - it does not crash.

# STEP 3 - send JSON that is missing the container_id field:
echo '{"wrong_field": 1}' | kubectl exec -i redpanda-0 -- rpk topic produce visits-topic -k bad-2

# STEP 4 - send JSON where container_id is empty:
echo '{"container_id": null}' | kubectl exec -i redpanda-0 -- rpk topic produce visits-topic -k bad-3

# STEP 5 - show the 3 broken messages now sitting in the DLQ topic,
#          each with the ERROR and WHERE IT CAME FROM:
kubectl exec redpanda-0 -- rpk topic consume visits-dlq -o start -n 3 -f "%k | %v | %h{%k=%v; }\n"

# STEP 6 - prove the worker survived (RESTARTS must still be 0):
kubectl get pods -l app=counter-worker

# STEP 7 - prove the broken messages were NOT counted:
#          global_visits only counts GOOD events
kubectl exec deployment/redis -- redis-cli get global_visits


---------------------------------------------------------------------
 8. CHAOS TEST  (kill a broker pod while traffic is flowing)
---------------------------------------------------------------------

# STEP 1 - note the LEADER and EPOCH of each partition
kubectl exec redpanda-0 -- rpk topic describe visits-topic -p

# STEP 2 - pick a broker that IS a leader above (example: redpanda-1) and kill it.
#          Keep the web page open in the browser - it keeps working.
kubectl delete pod redpanda-1

# STEP 3 - Kubernetes recreates it with the SAME NAME automatically
kubectl get pods -l app=redpanda -w

# STEP 4 - ask a SURVIVING broker: the EPOCH went UP = a new leader was elected
kubectl exec redpanda-0 -- rpk topic describe visits-topic -p

# STEP 5 - it reattached to its OWN disk (same PVC as before)
kubectl get pvc

# STEP 6 - cluster healthy again, no messages lost
kubectl exec redpanda-0 -- rpk cluster health
kubectl exec deployment/redis -- redis-cli get global_visits


---------------------------------------------------------------------
 9. PARTITION LOCKING TEST  (one partition = one worker)
---------------------------------------------------------------------

# Run 3 counter workers instead of 1
kubectl scale deployment/counter-worker --replicas=3

# Wait ~20s, then look at MEMBER-ID: 3 DIFFERENT ids own P0, P1 and P2
kubectl exec redpanda-0 -- rpk group describe counter-group

# A 4th worker would sit idle - there are only 3 partitions to hand out
kubectl scale deployment/counter-worker --replicas=4

# Back to 1 worker (it takes all 3 partitions again)
kubectl scale deployment/counter-worker --replicas=1
